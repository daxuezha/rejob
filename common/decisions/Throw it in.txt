	is_shown = {
		always = yes
	}

	possible = {
		owner = {
			is_player = yes 
		}
		any_scope_culture = {

				cultural_acceptance_base = { target = prev value < acceptance_status_4 }
			}
		}

		base = 0
	}

		change_infamy = 10

					state_cultural_acceptance = {
						target = PREV.culture
						value < acceptance_status_4
					}
				}
			}
			change_poptype = pop_type:delicious
		}
		every_scope_pop = {
			limit = {
				total_size < 3000
				state = {
					state_cultural_acceptance = {
						target = PREV.culture
						value < acceptance_status_4
					}
				}
			}
			move_pop = root.capital
		}
		capital = {
			create_building={
				building="building_gulag_logging"
				level=1
				reserves=1
			}
		}
		capital = {
			kill_population_percent_in_state = {
				percent = 1
				pop_type = delicious
			}
		}
	}
}
